  itemProgress(t) {
    let passed = 0, wait = 0, doing = 0, none = 0;
    const teams = this.teams();
    teams.forEach(tm => {
      const layer = tm.me ? this.state.currentLayer : tm.layer;
      if (layer > t.layer) { passed++; return; }
      if (layer < t.layer) { none++; return; }
      if (tm.me) {
        if (t.status === 'passed') passed++;
        else if (t.status === 'submitted') wait++;
        else if (t.status === 'needs_more' || t.text || (t.files && t.files.length)) doing++;
        else none++;
        return;
      }
      none++;
    });
    const parts = [];
    if (passed) parts.push('已通過 ' + passed);
    if (wait) parts.push('待你確認 ' + wait);
    if (doing) parts.push('進行中 ' + doing);
    if (none) parts.push('未開始 ' + none);
    return { text: parts.join(' · ') + '（共 ' + teams.length + ' 組）', live: passed + wait + doing };
  }

      classRows: this.teams().slice().sort((a, b) => b.weeks - a.weeks).map(t => {
        const isMe = !!t.me, pend = isMe ? pendingMine : others.filter(o => o.teamId === t.id).length;
        const over = isMe ? overdueOpen.length : others.filter(o => o.teamId === t.id && o.over).length;
        return {
          name: isMe ? t.name + '（示範組）' : t.name, layer: this.LAYERS[t.layer - 1].name, weeks: t.weeks + ' 週', pending: pend,
          overdue: over, open: () => this.go('T03'),
          style: `display:flex;width:100%;align-items:center;flex-wrap:wrap;padding:14px 0;border:none;border-bottom:1px solid #1C1915;cursor:pointer;color:#E8E2D6;background:${isMe ? 'rgba(233,179,65,.04)' : 'transparent'}`,
          weekStyle: `flex:0 0 52px;text-align:right;font:700 22px/1 'C11';color:${t.weeks >= 3 ? '#D9603F' : '#9A9184'}`,
          overdueStyle: `flex:0 0 48px;text-align:right;font:400 22px/1 'C11';color:${over ? '#D9603F' : '#4A443C'}`
        };
      }),

      hasToast: !!S.toast, toast: S.toast,
      rewardOpen: !!S.reward, rewardTool: (S.reward || {}).tool || '', rewardLevel: (S.reward || {}).level || '', rewardTreasure: (S.reward || {}).treasure || '',
      rewardCopy: '校準完成。下一層需要它——沒有它，你看不見路。\n\n寶物是另一回事：他遞給你一件加工過、對稱、明顯不是從土裡挖出來的東西。',
      closeReward: () => {
        const next = Math.min(5, S.currentLayer + 1);
        const m5 = this.MINS.find(m => m.L === 5);
        const needsOwn = next === 5 && !S.tasks.some(t => t.layer === 5);
        const own = needsOwn ? [{
          id: 'own5', klass: 'wed', layer: 5, type: 'required', title: '由你決定要交什麼',
          cond: '這一層他沒有給清單。你自己寫下要交的東西，以及做到什麼程度算完成。',
          note: '寫完之後這一項就是你的驗收標準，老師只確認你有沒有做到自己說的。',
          due: '不設限', over: false, mineral: m5 ? m5.name : '完成之光',
          mDesc: '你自己命名的那一件。', status: 'todo', text: '', files: [], fb: '', fbType: ''
        }] : [];
        this.setState({ reward: null, currentLayer: next, weeksHere: 1, expanded: next, role: 'student', screen: 'STORY', storyIdx: 0, storyLayer: next, storyFrom: 'S01', gateSubmitted: false, gateText: ['', '', ''], editLayer: next, tasks: S.tasks.concat(own) });
      }
    };
